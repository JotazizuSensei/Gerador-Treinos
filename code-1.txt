function videoMatch(v,cfg){
 const compat={
  'Mobilidade':['Body & Mind','Primal-inspired'],
  'Mobilidade + Core':['Mobilidade','Core','Functional Training'],
  'Body & Mind':['Mobilidade'],
  'Core':['Mobilidade + Core'],
  'Mix Training':['Functional Training'],
  'Functional Training':['Mix Training'],
  'Primal-inspired':['Mobilidade','Functional Training']
 };
 const exact=v.types.includes(cfg.classType);
 const adaptable=!exact&&(compat[cfg.classType]||[]).some(t=>v.types.includes(t));
 if(!exact&&!adaptable)return{score:0,reasons:[]};
 let score=exact?100:60,reasons=[exact?'modalidade':'adaptável'];
 const d=Math.abs(v.durationMin-cfg.duration);score-=Math.min(32,d*1.8);if(d<=6)reasons.push('duração');
 const dd=Math.abs(v.difficulty-cfg.difficulty);score-=dd*4;if(dd<=2)reasons.push('dificuldade');
 if(cfg.specialGroup!=='Geral'){
   if(v.special.includes(cfg.specialGroup)){score+=10;reasons.push('grupo')} else score-=12;
 }
 if(cfg.equipment==='Sem equipamento' && v.equipment!=='Sem equipamento')score-=14;
 if(cfg.equipment!=='Sem equipamento' && (v.equipment===cfg.equipment || lower(v.equipment).includes(lower(cfg.equipment))))score+=5;
 if(cfg.impact==='Baixo' && lower(v.impact).includes('médio'))score-=6;
 if(v.focus.includes(cfg.focus))score+=7;
 if(v.status==='EXTRAÍDO'){score+=exact?8:4;reasons.push('transcrição pronta')}
 return{score:Math.max(0,Math.min(100,Math.round(score))),reasons}
}
