function scoreEx(ex,phase,cfg,used){
 let s=prefScore(ex);const t=lower([ex.name,ex.region,ex.pattern,ex.objective,ex.position,ex.family].join(' ')),p=lower(phase);
 if(used.has(canon(ex)))s-=100;
 if(/aquec/.test(p)&&/circle|good morning|spinal|rotation|hip|mobility|reach/.test(t))s+=7;
 if(/core/.test(p)){if(ex.family==='Core')s+=14;else s-=5;if(/plank|dead bug|bridge|anti|hollow|bird dog|side plank/.test(t))s+=5}
 if(/mobil/.test(p)){if(ex.family==='Mobilidade'||ex.family==='Body & Mind')s+=10;else if(ex.family==='Core')s-=4}
 if(/flow|fundamental|principal/.test(p)&&/mobil|rotation|reach|wave|cars|lunge|squat|hinge|cossack/.test(t))s+=5;
 if((cfg.classType==='Mix Training'||cfg.classType==='Functional Training')&&/fundamental|principal|bloco|flow/.test(p)){if(ex.family==='Mix Training'||ex.family==='Functional Training')s+=8;if(ex.family==='Mobilidade')s-=4}
 if(/retorno|cool/.test(p)&&/twist|stretch|supine|child|happy|knee to chest|breath/.test(t))s+=7;
 if(cfg.focus==='Core'&&/core|plank|dead bug|bridge|anti/.test(t))s+=5;
 if(cfg.focus==='Anca / membros inferiores'&&/hip|anca|glute|lunge|squat|hamstring|ankle/.test(t))s+=5;
 if(cfg.focus==='Ombro / torácica'&&/shoulder|ombro|torac|scap|twist|reach|wrist/.test(t))s+=5;
 if(cfg.focus==='Cadeia posterior'&&/hinge|good morning|deadlift|bridge|hamstring|posterior/.test(t))s+=5;
 if(cfg.focus==='Coordenação / controlo'&&/switch|balance|rotation|sit through|single|locom/.test(t))s+=5;
 const lev=lower(ex.level+' '+ex.complexity);
 const exDif=/avanç|advanced/.test(lev)?8:/interm/.test(lev)?6:4;
 s-=Math.abs(exDif-cfg.difficulty)*1.8;
 if(cfg.difficulty<=2&&exDif>=7)s-=18;if(cfg.difficulty>=9&&exDif<=4)s-=5;
 return s
}
function pick(pool,phase,cfg,used,count){
 const ranked=pool.map(ex=>({ex,score:scoreEx(ex,phase,cfg,used)})).sort((a,b)=>b.score-a.score);
 const out=[];for(const r of ranked){if(out.length>=count)break;if(r.score<-30)continue;out.push(JSON.parse(JSON.stringify(r.ex)));used.add(canon(r.ex))}
 return out
}

function sortFlow(exs,cfg){if(cfg.flowRule==='Livre')return exs;return exs.slice().sort((a,b)=>positionRank(a.position)-positionRank(b.position))}
function flowCooldown(cfg,used,minutes,minRank){
 let pool=filterPool(APP.exercises.filter(x=>x.family==='Mobilidade'||x.family==='Body & Mind'),cfg);
 let strict=pool.filter(x=>positionRank(x.position)>=Math.max(4,Math.min(5,minRank??4)));
 if(strict.length<2)strict=pool.filter(x=>positionRank(x.position)>=4);
 let exs=pick(strict.length?strict:pool,'retorno à calma',cfg,used,2);
 exs=sortFlow(exs,cfg);setTimes(exs,minutes,50);return{phase:'Retorno à calma',minutes,exercises:exs}
}
function enforceGlobalFlow(blocks,cfg,used){
 if(cfg.flowRule!=='Prioritária')return blocks;
 const warm=blocks.filter(b=>/aquec/i.test(lower(b.phase)));
 const cool=blocks.filter(b=>/retorno|cool/i.test(lower(b.phase)));
 const middle=blocks.filter(b=>!/aquec|retorno|cool/i.test(lower(b.phase)));
 const middleMinutes=middle.reduce((a,b)=>a+(+b.minutes||0),0);
 let xs=middle.flatMap(b=>b.exercises||[]).slice().sort((a,b)=>positionRank(a.position)-positionRank(b.position));
 const groups=[];for(let r=0;r<=5;r++){const gx=xs.filter(x=>positionRank(x.position)===r);if(gx.length)groups.push({rank:r,exercises:gx})}
 const totalN=Math.max(1,xs.length);let allocated=0;
 const midBlocks=groups.map((g,i)=>{let m=i===groups.length-1?Math.round((middleMinutes-allocated)*10)/10:Math.round((middleMinutes*g.exercises.length/totalN)*10)/10;allocated+=m;const b={phase:'Parte fundamental — '+['Em pé','Meia altura','Joelhos / quadrupedia','Agachado','Sentado','Deitado'][g.rank],minutes:m,exercises:g.exercises};setTimes(b.exercises,b.minutes,30);return b});
 const coolMin=cool.reduce((a,b)=>a+(+b.minutes||0),0);
 const lastRank=groups.length?groups[groups.length-1].rank:0;
 const newCool=coolMin>0?[flowCooldown(cfg,used,coolMin,lastRank)]:[];
 return [...warm,...midBlocks,...newCool]
}
function scratch(){
 const cfg=getCfg(),used=new Set(),pool=filterPool(poolForType(cfg.classType),cfg),blocks=[warmupBlock(cfg)];
 let remaining=cfg.duration-blocks[0].minutes;
 if(cfg.classType==='Mobilidade + Core'){
   const mob=Math.round(remaining*.62),core=Math.round(remaining*.25),cool=remaining-mob-core;
   let a={phase:'Parte fundamental — Mobilidade',minutes:mob,exercises:sortFlow(pick(pool,'mobilidade',cfg,used,8),cfg)};setTimes(a.exercises,a.minutes);blocks.push(a);
   let c={phase:'Core / integração',minutes:core,exercises:sortFlow(pick(pool,'core',cfg,used,4),cfg)};setTimes(c.exercises,c.minutes);blocks.push(c);
   let co={phase:'Retorno à calma',minutes:cool,exercises:sortFlow(pick(pool,'retorno à calma',cfg,used,2),cfg)};setTimes(co.exercises,co.minutes,50);blocks.push(co)
 }else{
   const cool=Math.max(5,Math.round(remaining*.15)),main=remaining-cool;
   let a={phase:'Parte fundamental',minutes:main,exercises:sortFlow(pick(pool,'parte fundamental',cfg,used,Math.max(6,Math.round(main/4))),cfg)};setTimes(a.exercises,a.minutes);blocks.push(a);
   let co={phase:'Retorno à calma',minutes:cool,exercises:sortFlow(pick(pool,'retorno à calma',cfg,used,2),cfg)};setTimes(co.exercises,co.minutes,50);blocks.push(co)
 }
 const finalBlocks=enforceGlobalFlow(blocks,cfg,used);currentClass={id:'NEW-'+Date.now(),createdAt:new Date().toISOString(),config:cfg,mode:'NOVA',blocks:finalBlocks};finishClass();nav('build')
}
